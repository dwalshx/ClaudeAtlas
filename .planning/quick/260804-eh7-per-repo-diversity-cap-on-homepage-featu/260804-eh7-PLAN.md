mega-repo cannot dominate the visible shelf:

1. The Featured Skills shelf (`getFeaturedSkills(6)`) — cap to ONE card per repo.

       ```js
       // One Featured card per repo on the homepage shelf (diversity cap).
       // SELECTION-ONLY: does not touch scoring/tiering.
       export const MAX_PER_REPO_FEATURED = 1;

       /**
        * Diversity cap: from records already ordered best-first, take at most
        * `maxPerRepo` per repo until `limit` is reached, then BACKFILL with the
        * skipped records (original order) so the result is never short.
        * Returns exactly min(limit, records.length) items. Records missing
        * repo_full_name are bucketed by id (each its own unique bucket).
        * PURE — no scoring, no sorting, order-preserving.
        * @param {any[]} records
        * @param {number} maxPerRepo
        * @param {number} limit
        * @returns {any[]}
        */
       export function capPerRepo(records, maxPerRepo, limit) {
         if (!Array.isArray(records)) return [];
         const cap = Number.isFinite(maxPerRepo) && maxPerRepo > 0 ? maxPerRepo : Infinity;
         const target = Math.max(0, Math.min(limit, records.length));
         const counts = new Map();
         const picked = [];
         const skipped = [];
         for (const r of records) {
           if (picked.length >= target) { skipped.push(r); continue; }
           const key = (r && r.repo_full_name) || (r && r.id) || r;
           const n = counts.get(key) || 0;
           if (n < cap) {
             counts.set(key, n + 1);
             picked.push(r);
           } else {
             skipped.push(r);
           }
         }
         for (const r of skipped) {
           if (picked.length >= target) break;
           picked.push(r);
         }
         return picked;
       }
       ```
