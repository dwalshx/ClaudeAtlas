# scripts/lib/reciprocal-probe.js — PURE, zero I/O, zero imports, never throws (defensive on null/undefined inputs)

export const AGENT_UA = 'ClaudeAtlasBot/1.0 (+https://claudeatlas.com/bot)';
export const AGENT_TOKEN = 'ClaudeAtlasBot';                 // robots.txt User-agent token we match on
export const BROWSER_UA = 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/128.0.0.0 Safari/537.36';
export const WELL_KNOWN_PATHS = [                            // fixed probe set, in this order
  { path: '/llms.txt',                                      expect: 'text' },
  { path: '/llms-full.txt',                                 expect: 'text' },
  { path: '/.well-known/agents.json',                       expect: 'json' },
  { path: '/.well-known/mcp/server-card.json',              expect: 'json' },
  { path: '/.well-known/http-message-signatures-directory', expect: 'json' },   // "do THEY sign?"
  { path: '/ai.txt',                                        expect: 'text' },
];
export const AI_BOT_NAMES = ['GPTBot','ClaudeBot','Claude-Web','anthropic-ai','PerplexityBot','Google-Extended','Applebot-Extended','Bytespider','CCBot','meta-externalagent','Amazonbot','OAI-SearchBot','cohere-ai','Meta-ExternalFetcher','Diffbot'];
export const BODY_CAP_BYTES = 512 * 1024;
export const POLICY = { min_host_spacing_ms: 3000, concurrency: 3, timeout_ms: 15000, max_redirects: 5, retry_on_network_error: 1, method: 'GET' };

// robots.txt (RFC 9309 semantics, kept small)
export function parseRobots(text) -> { groups: Array<{ agents: string[], allow: string[], disallow: string[] }>, sitemaps: string[], content_signal: string|null }
//   - lines split on \n, strip '#' comments, trim; directives case-insensitive; consecutive User-agent lines share one group
//   - `Content-Signal:` directive captured verbatim (first occurrence)
export function selectGroup(robots, token = AGENT_TOKEN) -> group|null
//   - group whose agents include token (case-insensitive, exact token match) wins; else the '*' group; else null (= allow all)
export function isAllowed(robots, path, token = AGENT_TOKEN) -> boolean
//   - longest-match rule over the selected group's allow/disallow prefixes (support trailing '*' and '$' minimally); allow wins ties; empty `Disallow:` means allow; no group → true
export function analyzeRobots({ status, text, error }) -> {
  present: boolean, status: number|null, policy: 'parsed'|'allow_all'|'disallow_all',   // 2xx→parsed; 4xx→allow_all; 5xx/timeout/network→disallow_all (conservative)
  groups_count, sitemap_count, content_signal: string|null,
  ai_bots_named: string[], ai_bots_disallowed_all: string[],                              // subset of AI_BOT_NAMES appearing as User-agent / having Disallow: /
  disallows_all_bots: boolean,   // '*' group has Disallow: /
  disallows_us: boolean,         // isAllowed(robots,'/') === false
  mentions_claudeatlasbot: boolean,
  parsed: robots|null            // the parseRobots output, for the runner's per-path isAllowed checks
}

// response classification (pure function of status + headers + body snippet)
export function classifyResponse({ status, headers, bodySnippet, error }) -> 'allowed'|'blocked'|'challenged'|'toll'|'error'
//   error (timeout/network) → 'error'; 402 → 'toll'
//   'challenged' if headers['cf-mitigated'] === 'challenge' OR ((403|503) && body matches /Just a moment|cf-chl|Attention Required|captcha|Access Denied.{0,200}Reference #/is)
//   401/403/451/429 → 'blocked'; 2xx → 'allowed'; 3xx (redirect budget exhausted) and other 5xx → 'error'
export function summarizeResponse({ status, headers, bytes, bytesTruncated, redirectCount, finalUrl, bodySnippet, error }) -> {
  result, status, server, cf_ray: boolean, x_robots_tag, content_signal, content_type, bytes, bytes_truncated, redirect_count, final_url, final_host, error: string|null
}   // headers is a plain lowercase-keyed object (runner does Object.fromEntries(res.headers))
export function analyzeHtml(html) -> { jsonld_count: number, has_schema_org: boolean }
export function cloakingDiff(agent, browser) -> { differs: boolean, status_differs: boolean, size_ratio: number|null, reason: string|null }
//   differs = status differs OR |agent.bytes-browser.bytes| / max(browser.bytes,1) > 0.30; null/'error' on either side → differs:false, reason:'insufficient'
export function checkWellKnownBody(expect, contentType, bodySnippet) -> boolean   // 'json' → JSON.parse succeeds on snippet (or content-type json); 'text' → non-empty, not HTML (<!doctype|<html)
export function isMarkdownNegotiated(contentType) -> boolean                        // /^text\/markdown/i
export function aggregate(targets) -> {
  targets_total, by_result: { allowed, blocked, challenged, toll, error, robots_disallowed },
  non_error_count, block_rate, allowed_rate, challenged_count, toll_count,
  markdown_negotiation_rate, cloaking_count, jsonld_present_count, signers_count,
  standards: { [path]: { ok_count, rate } },
  robots: { present_count, names_ai_bots_count, disallows_all_bots_count, disallows_us_count, content_signal_count, has_sitemap_count },
  size_bytes: { min, median, max },
  by_tier: { [tier]: { n, allowed, blocked, challenged, toll, error, robots_disallowed, markdown_negotiated } },
  control: { domain: 'claudeatlas.com', allowed, markdown_negotiated, llms_txt_ok } | null,
  control_passes: boolean
}   // rates = fraction rounded to 3 dp; block_rate = (blocked+challenged)/non_error_count where non_error excludes 'error' AND 'robots_disallowed'; homepage-agent result is THE per-target result; toll = any 402 anywhere on the target
export function renderReport(pass) -> string   // markdown: headline numbers, per-tier table, per-target table (domain | tier | result | status | server | md-neg | llms.txt | signs | cloak | bytes), standards matrix (targets × WELL_KNOWN_PATHS as ✓/✗/—), robots posture table

# Per-target record written by the runner (all fields present, null when not applicable):
{ domain, operator, tier, source, note,
  robots: analyzeRobots(...) minus `parsed`,
  homepage: summarizeResponse(...) + { jsonld_count, has_schema_org },     // agent UA, Accept: text/html
  markdown: { result, status, content_type, negotiated, bytes },            // agent UA, Accept: text/markdown, text/html;q=0.9
  browser:  { result, status, bytes },                                      // BROWSER_UA, Accept: text/html
  cloaking: cloakingDiff(homepage, browser),
  well_known: { [path]: { result, status, content_type, bytes, plausible } },
  toll: boolean, signs_requests: boolean, requests_made: number, duration_ms: number }
# Any probe skipped by robots → { result: 'robots_disallowed', status: null, ... }.

# data/reciprocal-pass.json (schema_version 1, bounded ~27 × few KB):
{ schema_version: 1, run_at: ISO, agent: { ua: AGENT_UA, policy: POLICY, probe_paths: [...] }, targets: [...], aggregate: aggregate(targets) }
