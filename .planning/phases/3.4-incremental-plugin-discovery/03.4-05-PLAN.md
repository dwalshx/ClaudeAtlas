    Create .github/workflows/weekly-plugin-rewalk.yml — a LEAN workflow:
      name: Weekly Plugin Full Re-walk
      on:
        schedule:
          - cron: '0 2 * * 0'   # Sundays 02:00 UTC — finishes well before the 06:30 daily cron, which then picks up the fresh corpus
        workflow_dispatch: {}
      jobs:
        rewalk:
          runs-on: ubuntu-latest
          timeout-minutes: 360
          permissions:
            contents: read
          steps:
            - uses: actions/checkout@v4
            - uses: actions/setup-node@v4
              with: { node-version: 20, cache: npm }
            - run: npm ci
            - name: Restore plugins-raw cache
              uses: actions/cache/restore@v4
              with:
                path: data/plugins-raw.ndjson
                key: plugins-raw-ndjson-weekly-${{ github.run_number }}
                restore-keys: plugins-raw-ndjson-
            - name: Full re-walk (scrape-plugins, PLUGINS_FULL_REWALK=1)
              env:
                GITHUB_TOKEN: ${{ secrets.SCRAPE_PAT }}
                SCRAPE_PAT_CLASSIC: ${{ secrets.SCRAPE_PAT_CLASSIC }}
                PLUGINS_FULL_REWALK: '1'
              run: node scripts/scrape-plugins.js
            - name: Save plugins-raw cache
              if: always() && hashFiles('data/plugins-raw.ndjson') != ''
              uses: actions/cache/save@v4
              with:
                path: data/plugins-raw.ndjson
                key: plugins-raw-ndjson-weekly-${{ github.run_number }}
