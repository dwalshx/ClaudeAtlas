       const names = [...new Set(records.map(r => r.repo_full_name).filter(Boolean))];
       const fresh = new Map();
       const totalBatches = Math.ceil(names.length / REFRESH_BATCH_SIZE);
       let bi = 0;
       for (let i = 0; i < names.length; i += REFRESH_BATCH_SIZE) {
