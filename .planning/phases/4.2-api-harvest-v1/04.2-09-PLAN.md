```js
const specStats = { fetched: 0, reused: 0, capped: 0, failed: 0, skipped: 0 };
const guru = records.filter((r) => r.origin.source === 'apis.guru');
if (mode === 'catalog-only') { for (const rec of guru) { markSpecUnparsed(rec, 'catalog_only'); specStats.skipped++; } }
else if (listReused) { specStats.reused = guru.length; }                    // whole-source reuse on 304 (prior records already carry spec fields + access + id)
else {
  let processed = 0;
  await Promise.all(guru.map((rec) => pool.run(rec.extra.spec_url ?? LIST_URL, async () => {
    const p = prior.get(rawKey(rec));
    if (p && p.raw_source?.updated === rec.raw_source.updated && p.extra?.spec_parsed === true) {
      // per-record reuse: copy the spec-derived fields + the merged access from the prior record
      rec.extra = { ...rec.extra, base_urls: p.extra.base_urls, spec_version: p.extra.spec_version, endpoint_count: p.extra.endpoint_count, endpoint_summaries: p.extra.endpoint_summaries ?? [], spec_parsed: true };
      if (p.docs_url && rec.docs_url == null) rec.docs_url = p.docs_url;
      rec.access = structuredClone(p.access); specStats.reused++;
    } else if (!rec.extra.spec_url) { markSpecUnparsed(rec, 'no_spec_url'); specStats.skipped++; }
    else {
      try {
        const r = await fetchWithRetry(rec.extra.spec_url, { ...fetchOpts, maxBytes: SPEC_MAX_BYTES, headers: { Accept: 'application/json' } });
        if (r.capped) { markSpecUnparsed(rec, 'capped'); specStats.capped++; }
        else if (!r.ok || !r.body) { markSpecUnparsed(rec, `http_${r.status}`); specStats.failed++; }
        else {
          let spec = null; try { spec = JSON.parse(r.body); } catch { spec = null; }
          const ex = extractOpenApi(spec);
          if (!ex) { markSpecUnparsed(rec, 'invalid_json'); specStats.failed++; }
          else { applySpec(rec, ex); rec.access = mergeAccess(rec.access, accessFromSpec(ex, rec.extra.spec_url, now())); specStats.fetched++; }
        }
      } catch { markSpecUnparsed(rec, 'fetch_error'); specStats.failed++; }
    }
    processed++;
    if (checkpointEvery > 0 && processed % checkpointEvery === 0) writeNdjsonStreaming(partialPath ?? PARTIAL_PATH, records, { header: buildHeader('api') });
  })));
}
log(`[harvest-apis] spec pass: fetched=${specStats.fetched} reused=${specStats.reused} capped=${specStats.capped} failed=${specStats.failed} skipped=${specStats.skipped} in ${specMs} ms`);
```
