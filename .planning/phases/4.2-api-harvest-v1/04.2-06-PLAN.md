export const METHODS = new Set(['get', 'put', 'post', 'delete', 'options', 'head', 'patch', 'trace']);
function fillServerVars(s) {
  let u = typeof s?.url === 'string' ? s.url : '';
  for (const [k, v] of Object.entries(s?.variables || {})) u = u.replaceAll(`{${k}}`, v?.default != null ? String(v.default) : '');
  return u || null;
}
export function extractOpenApi(spec) {
  if (!spec || typeof spec !== 'object' || Array.isArray(spec)) return null;
  const isV2 = typeof spec.swagger === 'string';
  const base_urls = isV2
    ? (spec.host ? [`${(Array.isArray(spec.schemes) && spec.schemes[0]) || 'https'}://${spec.host}${spec.basePath || ''}`] : [])
    : (Array.isArray(spec.servers) ? spec.servers.map(fillServerVars).filter(Boolean) : []);
  const schemes = (isV2 ? spec.securityDefinitions : spec.components?.securitySchemes) || {};
  const media = []; const seen = new Set(); const add = (mt) => { if (typeof mt === 'string' && !seen.has(mt)) { seen.add(mt); media.push(mt); } };
  const summaries = []; let ops = 0; let hasGraphql = false;
  for (const [p, item] of Object.entries(spec.paths || {})) {
    if (/\/graphql\b/i.test(p)) hasGraphql = true;
    if (!item || typeof item !== 'object') continue;
    for (const [m, op] of Object.entries(item)) {
      if (!METHODS.has(m)) continue;
      ops++;
      if (summaries.length < 10) summaries.push(`${m.toUpperCase()} ${p} ${op?.summary || op?.operationId || ''}`.trim().slice(0, 80));
      for (const r of Object.values(op?.responses || {})) for (const mt of Object.keys(r?.content || {})) add(mt);
      for (const mt of op?.produces || []) add(mt);
    }
  }
  for (const mt of spec.produces || []) add(mt);
  return {
    spec_version: spec.openapi || spec.swagger || null, base_urls, endpoint_count: ops,
    security: { schemes, requirements: Array.isArray(spec.security) ? spec.security : [] },
    media_types: media, license: spec.info?.license ?? null, terms: spec.info?.termsOfService ?? null,
    external_docs_url: spec.externalDocs?.url ?? null, transport: hasGraphql ? 'graphql' : 'rest',
    endpoint_summaries: summaries, info_title: spec.info?.title ?? null, info_description: spec.info?.description ?? null,
  };
}
