```js
export const DEFAULT_ACCEPT = 'application/json, text/csv, text/plain;q=0.9, */*;q=0.1';

export async function readBodyCapped(res, maxBytes = Infinity) {
  if (!res.body || typeof res.body.getReader !== 'function') {            // non-stream Response (tests may pass plain bodies)
    const text = await res.text(); const bytes = Buffer.byteLength(text);
    return bytes > maxBytes ? { capped: true, body: null, bytes } : { capped: false, body: text, bytes };
  }
  const reader = res.body.getReader(); const chunks = []; let bytes = 0;
  for (;;) {
    const { done, value } = await reader.read(); if (done) break;
    bytes += value.byteLength;
    if (bytes > maxBytes) { try { await reader.cancel(); } catch {} return { capped: true, body: null, bytes }; }
    chunks.push(value);
  }
  return { capped: false, body: Buffer.concat(chunks.map((c) => Buffer.from(c))).toString('utf-8'), bytes };
}

export async function politeFetch(url, { fetchImpl = fetch, etagCache, timeoutMs = 20000, maxBytes = Infinity, headers = {}, now = () => new Date().toISOString() } = {}) {
  const prior = etagCache?.get(url);
  const h = { 'User-Agent': AGENT_UA, Accept: DEFAULT_ACCEPT, ...headers };
  if (prior?.etag) h['If-None-Match'] = prior.etag;
  else if (prior?.last_modified) h['If-Modified-Since'] = prior.last_modified;
  const res = await fetchImpl(url, { headers: h, signal: AbortSignal.timeout(timeoutMs), redirect: 'follow' });
  const out = { status: res.status, ok: res.ok, notModified: res.status === 304, capped: false, body: null,
    headers: { etag: res.headers.get('etag'), lastModified: res.headers.get('last-modified'), contentType: res.headers.get('content-type'), retryAfter: res.headers.get('retry-after') } };
  if (out.notModified) return out;
  const { capped, body } = await readBodyCapped(res, maxBytes);
  out.capped = capped; out.body = body;
  if (res.ok && (out.headers.etag || out.headers.lastModified) && etagCache) {
    etagCache.set(url, { url, etag: out.headers.etag ?? null, last_modified: out.headers.lastModified ?? null, checked_at: now(), status: res.status });
  }
  return out;
}

export class EtagCache {
  constructor(map = new Map()) { this.map = map; }
  static load(path) { return new EtagCache(readNdjsonRecords(path, { keyFn: (r) => r.url })); }   // missing file → empty Map (helper semantics)
  get(url) { return this.map.get(url); }
  set(url, entry) { this.map.set(url, { ...entry, url }); }
  delete(url) { this.map.delete(url); }
  get size() { return this.map.size; }
  save(path) { writeNdjsonStreaming(path, [...this.map.values()]); }   // no header: this is a sidecar, not an entity file
}
```

```js
const defaultSleep = (ms) => new Promise((r) => setTimeout(r, ms));
function retryDelay(res, attempt, backoffMs) {
  const ra = res?.headers?.retryAfter ?? res?.headers?.get?.('retry-after');
  const secs = ra != null ? Number(ra) : NaN;
  if (Number.isFinite(secs) && secs >= 0) return Math.min(secs * 1000, 120000);
  return backoffMs[Math.min(attempt, backoffMs.length - 1)];
}
export async function fetchWithRetry(url, { retries = 3, backoffMs = [1000, 4000, 16000], sleep = defaultSleep, ...opts } = {}) {
  let attempt = 0; let lastErr = null; let res = null;
  for (;;) {
    try {
      res = await politeFetch(url, opts);
      if (!(res.status === 429 || res.status >= 500)) return res;          // success, 304, or a non-retryable 4xx
    } catch (err) {
      if (!(err instanceof TypeError || err?.name === 'AbortError' || err?.name === 'TimeoutError')) throw err;
      lastErr = err; res = null;
    }
    if (attempt >= retries) { if (res) return res; throw lastErr; }
    await sleep(retryDelay(res, attempt, backoffMs));
    attempt++;
  }
}
export class FetchPool {
  constructor({ limit = 4, perHostMs = 250, now = Date.now, sleep = defaultSleep } = {}) {
    this.limit = limit; this.perHostMs = perHostMs; this.now = now; this.sleep = sleep;
    this.active = 0; this.queue = []; this.lastStart = new Map();   // host → timestamp of the last start
  }
  get inFlight() { return this.active; }
  run(url, fn) {
    return new Promise((resolve, reject) => { this.queue.push({ url, fn, resolve, reject }); this._pump(); });
  }
  async _pump() {
    while (this.active < this.limit && this.queue.length) {
      const job = this.queue.shift(); this.active++;
      this._exec(job).finally(() => { this.active--; this._pump(); });
    }
  }
  async _exec({ url, fn, resolve, reject }) {
    const host = hostOf(url) || '*';
    const last = this.lastStart.get(host);
    const wait = last == null ? 0 : Math.max(0, last + this.perHostMs - this.now());
    if (wait > 0) await this.sleep(wait);
    this.lastStart.set(host, this.now());
    try { resolve(await fn()); } catch (e) { reject(e); }
  }
}
```
