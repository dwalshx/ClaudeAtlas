```js
export const SOURCE_PRECEDENCE = ['apis.guru', 'data.gouv.fr', 'uk-api-catalogue', 'public-apis'];
export const STRIP_LABELS = ['www', 'api', 'apis', 'developer', 'developers', 'dev', 'docs', 'open', 'opendata', 'data', 'portal', 'sandbox', 'public'];
export function canonicalHost(url) {
  const h = hostOf(url).toLowerCase(); if (!h) return '';
  const labels = h.split('.');
  if (labels.length > 2 && STRIP_LABELS.includes(labels[0])) return labels.slice(1).join('.');
  return h;
}
export function nameStem(name) {
  const s = slugify(name); let t = s.replace(/^api-/, '').replace(/-api$/, ''); return t || s;
}
export function mergeKey(rec) {
  const primary = rec?.extra?.base_urls?.[0] || rec?.home_url || rec?.docs_url || '';
  return `${canonicalHost(primary)}/${nameStem(rec?.name ?? '')}`;
}
```

```js
export const FILL_ACCESS_PATHS = ['auth.type', 'auth.schemes', 'cost.model', 'cost.free_tier', 'rate_limit.requests', 'rate_limit.per', 'rate_limit.notes', 'license.spdx', 'license.name', 'license.url', 'formats', 'transport'];
const isUnknown = (v) => v == null || v === '' || v === 'unknown' || (Array.isArray(v) && v.length === 0);
const getPath = (obj, path) => path.split('.').reduce((o, k) => (o == null ? undefined : o[k]), obj);
function setPath(obj, path, value) { const ks = path.split('.'); let o = obj; for (const k of ks.slice(0, -1)) { if (!o[k] || typeof o[k] !== 'object') o[k] = {}; o = o[k]; } o[ks[ks.length - 1]] = value; }
const rank = (rec) => { const i = SOURCE_PRECEDENCE.indexOf(rec?.origin?.source); return i === -1 ? SOURCE_PRECEDENCE.length : i; };
function cmp(a, b) {
  const r = rank(a) - rank(b); if (r) return r;
  const am = a?.extra?.source_modified_at ?? '', bm = b?.extra?.source_modified_at ?? '';
  if (am !== bm) return am === '' ? 1 : bm === '' ? -1 : (am > bm ? -1 : 1);      // newest first, null last
  return String(a?.origin?.source_id ?? '').localeCompare(String(b?.origin?.source_id ?? ''));
}
function absorb(winner, loser) {
  const ws = winner.origin.source, ls = loser?.origin?.source;
  if (ls && ls !== ws && !winner.origin.also_listed_in.includes(ls)) winner.origin.also_listed_in.push(ls);
  if (!winner.access || typeof winner.access !== 'object') winner.access = {};
  if (!winner.access.provenance) winner.access.provenance = {};
  for (const path of FILL_ACCESS_PATHS) {
    const wv = getPath(winner.access, path), lv = getPath(loser?.access, path);
    if (isUnknown(wv) && !isUnknown(lv)) { setPath(winner.access, path, structuredClone(lv)); const p = loser?.access?.provenance?.[path]; if (p) winner.access.provenance[path] = structuredClone(p); }
  }
  if (winner.docs_url == null && loser?.docs_url) winner.docs_url = loser.docs_url;
  if (winner.extra.spec_url == null && loser?.extra?.spec_url) winner.extra.spec_url = loser.extra.spec_url;
  if ((winner.description ?? '') === '' && typeof loser?.description === 'string' && loser.description) winner.description = loser.description;   // decision 6: fill only when EMPTY
  const cats = new Set(winner.extra.categories_src || []); for (const c of loser?.extra?.categories_src || []) cats.add(c); winner.extra.categories_src = [...cats];
}
export function mergeApiRecords(records) {
  const groups = new Map();
  for (const r of records || []) { const k = mergeKey(r); if (!groups.has(k)) groups.set(k, []); groups.get(k).push(r); }
  const merged = []; let cross = 0, same = 0;
  for (const group of groups.values()) {
    const sorted = [...group].sort(cmp);
    const winner = structuredClone(sorted[0]);
    for (const loser of sorted.slice(1)) { if (loser.origin?.source === winner.origin.source) same++; else cross++; absorb(winner, loser); }
    merged.push(winner);
  }
  return { merged, stats: { groups: groups.size, merged_duplicates: { cross_source: cross, same_source: same } } };
}
```
