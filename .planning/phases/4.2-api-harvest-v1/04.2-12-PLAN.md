```yaml
name: Measure Data Feeds (Phase 4.2, branch-only)

# Phase 4.2 Plan 12 — measures the api legs INDEPENDENTLY of daily-scrape.yml
# (main's cron has been red since 2026-09-28 — Research Pitfall 11 — so a
# branch dispatch of daily-scrape.yml dies before the api steps). Runs

# the SAME prefixes as daily-scrape.yml (apis-raw-ndjson-, api-vectors-) so a
# later daily run on this branch restores them. Delete or keep after the flip
# (3.4's weekly workflow precedent — it is harmless on main because it is
# dispatch-only).

on:
  workflow_dispatch:
    inputs:
      live_embed:
        description: 'Use the real OPENAI_API_KEY (one-time live smoke, ~$0.01). Default: EMBED_DRY_RUN=1 deterministic fake vectors.'
        type: boolean
        default: false
      limit:
        description: 'Optional --limit N for harvest-apis.js (smoke); leave empty for the full run'
        type: string
        default: ''

jobs:
  measure:
    runs-on: ubuntu-latest
    timeout-minutes: 90
    permissions:
      contents: read
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: npm
      - run: npm ci

      - name: Restore apis-raw cache
        uses: actions/cache/restore@v4
        with:
          path: |
            data/apis-raw.ndjson
            data/feeds-etag-cache.ndjson
          key: apis-raw-ndjson-measure-${{ github.run_number }}
          restore-keys: |
            apis-raw-ndjson-

      - name: Harvest APIs (timed)
        run: |
          start=$(date +%s)
          if [ -n "${{ inputs.limit }}" ]; then node scripts/harvest-apis.js --limit "${{ inputs.limit }}"; else node scripts/harvest-apis.js; fi
          echo "::notice title=harvest-apis::$(( $(date +%s) - start )) s"
          echo "raw records: $(( $(wc -l < data/apis-raw.ndjson) - 1 ))"

      - name: Save apis-raw cache
        if: ${{ always() && hashFiles('data/apis-raw.ndjson') != '' }}
        uses: actions/cache/save@v4
        with:
          path: |
            data/apis-raw.ndjson
            data/feeds-etag-cache.ndjson
          key: apis-raw-ndjson-measure-${{ github.run_number }}

      - name: Filter APIs (timed)
        run: |
          start=$(date +%s)
          node scripts/filter-apis.js
          echo "::notice title=filter-apis::$(( $(date +%s) - start )) s"
          echo "api records: $(( $(wc -l < data/apis.ndjson) - 1 ))"

      - name: Restore api-vectors cache
        uses: actions/cache/restore@v4
        with:
          path: data/api-vectors.ndjson
          key: api-vectors-measure-${{ github.run_number }}
          restore-keys: api-vectors-

      - name: Embed APIs (timed; dry-run unless live_embed)
        env:
          OPENAI_API_KEY: ${{ secrets.OPENAI_API_KEY }}
          # Decision 1: live only when the dispatch input is true AND the secret exists; otherwise deterministic fake vectors.
          EMBED_DRY_RUN: ${{ (inputs.live_embed == true && secrets.OPENAI_API_KEY != '') && '0' || '1' }}
        run: |
          start=$(date +%s)
          echo "EMBED_DRY_RUN=$EMBED_DRY_RUN"
          node scripts/embed-skills.js --entity-type api
          echo "::notice title=embed-api::$(( $(date +%s) - start )) s"

      - name: Save api-vectors cache
        if: ${{ always() && hashFiles('data/api-vectors.ndjson') != '' }}
        uses: actions/cache/save@v4
        with:
          path: data/api-vectors.ndjson
          key: api-vectors-measure-${{ github.run_number }}

      - name: Enrich APIs (timed)
        run: |
          start=$(date +%s)
          node scripts/enrich.js --input data/api-vectors.ndjson --records data/apis.ndjson --entity-type api
          echo "::notice title=enrich-api::$(( $(date +%s) - start )) s"

      - name: Upload vectors DRY RUN (validates payload; no Vectorize calls)
        run: |
          start=$(date +%s)
          node scripts/upload-vectors.js --input data/api-vectors.ndjson --dry-run
          echo "::notice title=upload-dry-run::$(( $(date +%s) - start )) s"

      - name: Summary
        run: |
          {
            echo "## Phase 4.2 api legs — measured"
            echo "raw: $(( $(wc -l < data/apis-raw.ndjson) - 1 ))  filtered: $(( $(wc -l < data/apis.ndjson) - 1 ))  vectors: $(( $(wc -l < data/api-vectors.ndjson) - 1 ))"
            echo "EMBED_DRY_RUN=${{ (inputs.live_embed == true && secrets.OPENAI_API_KEY != '') && '0' || '1' }}"
          } >> "$GITHUB_STEP_SUMMARY"
```
