## 1. The concrete EntityRecord shape

JavaScript project, so we ship the canonical shape as **JSDoc typedefs** in
`src/lib/types.js` (consumed by VS Code + Astro's TS-language-server) **plus** a
companion `src/lib/types.d.ts` for tooling that prefers ambient `.d.ts`. Both files
generated from a single source-of-truth comment block to prevent drift; a CI lint
diffs them.

```typescript
// src/lib/types.d.ts — canonical

export type EntityType =
  | 'skill'
  | 'plugin'
  | 'mcp_server'
  | 'command_lib'
  | 'agent_lib'
  | 'hook_lib';
// NOTE: 'framework' is intentionally NOT an entity_type (Q3 decision: frameworks
// are tag-based, e.g. `framework:gsd` attachable to any entity).

export type QualityTier = 'featured' | 'solid' | 'listed';
// Tier rename to 'top' / 'solid' / 'indexed' is Phase 3.6 — not in F2.

export interface EntityCommon {
  // --- Identity ---
  id: string;                              // `${entity_type}:${repo_full_name}/${path_within_repo}`
  slug: string;                            // URL-safe; scoped per type → `/{entity_type}s/{owner}/{name}/`
  entity_type: EntityType;
  name: string;
  description: string;

  // --- Repo metadata (every entity lives in a GitHub repo) ---
  repo_full_name: string;
  repo_url: string;
  repo_stars: number;
  repo_forks: number;
  repo_open_issues: number;
  repo_topics: string[];
  repo_license: string | null;
  repo_language: string | null;
  repo_created_at: string;     // ISO 8601
  repo_updated_at: string;
  repo_pushed_at: string;
  repo_owner_type: 'Organization' | 'User';
  repo_owner_avatar: string;
  repo_archived: boolean;
  repo_is_fork: boolean;
  repo_description: string | null;

  // --- Content (every entity has SOME text) ---
  body_length: number;
  // NOTE: body_markdown lives in `extra.*` because what counts as "primary content"
  // is entity-type-shaped (SKILL.md for skills, plugin.json + README for plugins, etc.)

  // --- Computed (filled by scorer + filter) ---
  quality_score: number;       // 0-100, computed by per-type scorer
  quality_tier: QualityTier;
  novelty_score: number;       // 0-1; populated by enrich.js (lands in F3); 0 in F2
  is_duplicate: boolean;       // populated by enrich.js; false in F2
  canonical_id: string | null; // points to canonical EntityRecord.id; null in F2

  // --- Classification ---
  tags: string[];              // PRIMARY classifier; conventions in scripts/lib/tags.js
  category: string | null;     // LEGACY display field; mechanically equals tags.find(t => t.startsWith('category:')).split(':')[1]
                               // Kept for back-compat; removed in 3.6.

  // --- Lineage ---
  scraped_at: string;
  content_sha: string;
  source: 'code-search' | 'topics' | 'seed' | 'discover' | 'registry' | 'manual';
  discovery_signals: string[]; // which discovery signals matched (audit trail)
  schema_version: 2;           // bumped from implicit 1
}

export interface SkillExtra {
  type: 'skill';
  skill_path: string;                         // path within repo (e.g. "skills/foo/SKILL.md")
  body_markdown: string;                      // first 1500 chars (filter) or 5000 chars (raw)
  frontmatter: Record<string, any>;
  has_name: boolean;
  has_description: boolean;
  skill_first_commit_at: string | null;
}

export interface PluginExtra {
  type: 'plugin';
  body_markdown: string;                      // README excerpt
  manifest: {
    name: string;
    description: string;
    version: string;
    author: { name: string; email?: string } | null;
  };
  components: {
    skills: number;
    agents: number;
    commands: number;
    hooks: number;
    mcp_servers: number;
    lsp_servers: number;
    total: number;
  };
  component_list: Array<{
    type: 'skill' | 'agent' | 'command' | 'hook' | 'mcp_server' | 'lsp_server';
    name: string;
    path: string;
    description?: string;
  }>;
  from_marketplace: string | null;            // marketplace owner/repo if discovered via marketplace.json
  marketplace_manifest: Record<string, any> | null;
  install_command: string;                    // `claude plugin install <repo>` form
}

export interface McpExtra {
  type: 'mcp_server';
  body_markdown: string;
  transport: 'stdio' | 'sse' | 'http';
  capabilities: Array<'tools' | 'resources' | 'prompts'>;
  registry_id: string | null;
  install_command: string;
}

export interface CommandLibExtra {
  type: 'command_lib';
  body_markdown: string;
  commands_dir: string;                       // e.g. ".claude/commands"
  command_count: number;
  command_names: string[];
}

export interface AgentLibExtra {
  type: 'agent_lib';
  body_markdown: string;
  agents_dir: string;
  agent_count: number;
  agent_names: string[];
}

export interface HookLibExtra {
  type: 'hook_lib';
  body_markdown: string;
  hooks_dir: string;
  hook_definitions: Array<{ event: string; command_path: string }>;
}

export type EntityRecord =
  | (EntityCommon & { entity_type: 'skill';       extra: SkillExtra })
  | (EntityCommon & { entity_type: 'plugin';      extra: PluginExtra })
  | (EntityCommon & { entity_type: 'mcp_server';  extra: McpExtra })
  | (EntityCommon & { entity_type: 'command_lib'; extra: CommandLibExtra })
  | (EntityCommon & { entity_type: 'agent_lib';   extra: AgentLibExtra })
  | (EntityCommon & { entity_type: 'hook_lib';    extra: HookLibExtra });
```

**Notes / design choices:**

- `body_length` lives in common (asked for everywhere); `body_markdown` lives in
  `extra.*` because per-entity-type input rules differ. The dup is intentional —
  `body_length` is a search/scorer signal; `body_markdown` is content.
- `category` is kept as a derived legacy field for back-compat through F2 → 3.6
  (User Q3 decision: tags primary, category goes away). Setting `category` from
  `tags` happens in `scripts/lib/tags.js::projectCategoryFromTags()` after tag
  derivation. Pages reading `entity.category` continue to work.
- `tags` is the new primary axis. Conventions documented in §5.
- `discovery_signals` is new; populated by the recipe.
- `schema_version: 2` is explicit on every record; the legacy reader stamps it on
  the fly when upcasting v1 → v2 (§7).
- Per-entity-type `body_markdown` truncation budgets stay where they are today
  (filter trims to 1500 chars for committed `skills.ndjson`; scraper writes 5000
  chars to `skills-raw.ndjson`). No change.

`src/lib/types.js` ships the same shape as JSDoc `@typedef` blocks that Astro and
VS Code consume directly without a TypeScript compile step.

## 3. The recipe + engine architecture

Recipe interface (JSDoc-typed):

```typescript
// scripts/lib/recipe-engine.js

export interface DiscoveryHit {
  repo_full_name: string;
  path: string;             // path within repo to the entity's primary file/dir
  signals: string[];        // which discovery signals matched (audit)
  source: 'code-search' | 'topics' | 'seed' | 'registry' | 'manual';
  // Recipe-specific blob the engine doesn't touch
  raw?: Record<string, any>;
}

export interface DiscoveryRecipe<Extra> {
  entity_type: EntityType;
  output_path: string;                       // e.g. 'data/skills-raw.ndjson'
  state_path: string;                        // e.g. 'data/scrape-skills-state.json'

  // Discovery: emits hits. Engine handles fan-in, dedup, checkpointing.
  discover(opts: { mode: 'full' | 'incremental'; since?: string }):
    AsyncIterable<DiscoveryHit>;

  // Parse a single hit into an EntityRecord. Engine handles HTTP, ETag,
  // rate-limit, retry around the inner fetch calls via the passed `fetch`
  // closure (which is the engine's rate-limited+ETag-cached fetch).
  parse(
    hit: DiscoveryHit,
    ctx: { fetch: typeof fetch; repoMeta: GitHubRepo },
  ): Promise<EntityRecord<Extra> | null>;    // null = filtered out at parse stage

  // Determine the canonical id for a hit (before fetch) so the engine can
  // implement skip-known cheaply.
  computeId(hit: DiscoveryHit): string;

  // Optional: per-recipe cleanup or assertion
  postProcess?(record: EntityRecord<Extra>): EntityRecord<Extra>;
}
```

Engine responsibilities (NOT recipe responsibilities):
- ETag caching (`data/etag-cache.json` via F1 chunked-read helper)
- Rate limiting (search 9 req/min, general 4800 req/hr — same as today's
  `rateLimitedFetch`)
- Checkpointing every N records via F1's NDJSON `writeNdjsonStreaming` append mode
- Resumable state in `state_path`
- Skip-known via `content_sha` lookup
- Logging + metrics emission

Recipe responsibilities:
- Discovery query construction (GitHub code search OR repo search OR registry pull)
- Parse one hit → one EntityRecord
- Compute id

`skill.recipe.js` is constructed by **moving** (not copying) the relevant code
out of today's `scrape.js`: `discover()` wraps the code-search + topics + seed
+ tree-walk passes; `parse()` wraps `parseSkill()` + `categorizeSkill()` +
`scoreSkill()` calls; `computeId()` returns `skill:${repo}/${path}`.

**Hypothetical plugin recipe trace (validation that the interface fits 3.2):**

```javascript
// scripts/lib/recipes/plugin.recipe.js — 3.2, not F2

import { parsePluginManifest, walkComponents } from '../plugin-parsers.js';

export const pluginRecipe = {
  entity_type: 'plugin',
  output_path: 'data/plugins-raw.ndjson',
  state_path: 'data/scrape-plugins-state.json',

  async *discover({ mode, since }) {
    // 1. Code-search for filename:plugin.json path:.claude-plugin
    for (const hit of searchCode('filename:plugin.json path:.claude-plugin')) {
      yield { repo_full_name: hit.repo, path: hit.path, signals: ['plugin_json'], source: 'code-search' };
    }
    // 2. Code-search for filename:marketplace.json
    for (const hit of searchCode('filename:marketplace.json path:.claude-plugin')) {
      yield { repo_full_name: hit.repo, path: hit.path, signals: ['marketplace_json'], source: 'code-search' };
    }
    // 3. For each marketplace, walk it for source URLs
    // (handled by recipe internals)
  },

  computeId(hit) {
    return `plugin:${hit.repo_full_name}/${hit.path.replace(/\.claude-plugin\/(plugin|marketplace)\.json$/, '')}`;
  },

  async parse(hit, { fetch, repoMeta }) {
    const manifestJson = await fetch(...);
    const manifest = parsePluginManifest(manifestJson);
    if (!manifest) return null;
    const components = await walkComponents(hit.repo_full_name, hit.path, fetch);

    const common = buildCommonFromRepoMeta(repoMeta, hit, /* entity_type */ 'plugin');
    const extra = {
      type: 'plugin',
      body_markdown: '', // pulled from repo README in a separate engine call
      manifest,
      components: components.summary,
      component_list: components.list,
      from_marketplace: hit.raw?.marketplace_source ?? null,
      marketplace_manifest: hit.raw?.marketplace_manifest ?? null,
      install_command: `claude plugin install ${hit.repo_full_name}`,
    };

    return { ...common, entity_type: 'plugin', extra };
  },
};
```

## 5. Tag schema

**Format:** lowercase ASCII strings, hyphenated for word boundaries, with
optional namespace prefix using `:` as separator. Examples:

- Unnamespaced: `testing`, `cli`, `python` — free-form taxonomic tags
- Namespaced: `category:ai-automation`, `framework:gsd`, `language:python`,
  `topic:claude-code`, `mcp:tools`, `plugin:components-gt-5`

**Conventions (documented in `scripts/lib/tags.js` and `methodology.astro`):**

| Namespace | Source | Cardinality |
|-----------|--------|-------------|
| `category:` | Mechanical migration from legacy `category` field (F2) + future classifier (3.4+) | ~8 today, ~12 at saturation |
| `framework:` | README badge detection + repo topic heuristics + curated seed list (3.4+) | ~15 known frameworks |
| `language:` | `repo_language` field, lowercased | ~30 |
| `topic:` | `repo_topics[]` whitelisted to ~80 known-good topics | ~80 |
| (none) | Free-form: SKILL.md frontmatter `tags:` array, README inference (3.4+), embedding cluster membership | Unbounded — capped at top-N per entity |

**F2-mechanical population (the only tag source enabled in this phase):**

```javascript
// scripts/lib/tags.js
export function deriveTagsFromLegacyCategory(category) {
  if (!category) return [];
  const slug = category.toLowerCase().replace(/[&]/g, 'and').replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '');
  return [`category:${slug}`];
}

export function projectCategoryFromTags(tags) {
  const t = tags.find(t => t.startsWith('category:'));
  if (!t) return null;
  // Reverse the slug → display name lookup via a fixed table (8 entries today).
  return CATEGORY_DISPLAY[t.slice('category:'.length)] || null;
}
```

**Future tag sources (NOT implemented in F2; schema accommodates them):**

- Frontmatter declared tags (already parsed by `parse-skill.js`, today populates
  `record.tags` directly; in F2 we route them through `validateTag()` and merge
  with category-derived tags). The merge logic lands in F2 because today
  `frontmatter.tags` is dropped on the floor for some skills — fixing it is
  trivial inside the migration and worth doing once.
- Repo topics → topic tags (3.4+).
- README inference / classifier output (3.4+).
- Embedding cluster membership (3.4+, using F3's enrich output).

## 6. Cross-type search ranking — v1 formula

**Acknowledged uncertainty:** there are zero non-skill records in the catalog at
F2-land time. The ranking is correct by construction for single-type results
(behavior matches today's quality_score sort). Cross-type ranking needs
empirical calibration starting the day after 3.2 ships and plugin records land
in the index. **F2 ships a defensible v1; 3.5 planning is the calibration
checkpoint.**

```javascript
// worker/index.js — display_score computed per match before sort

function displayScore(match, query) {
  const semantic = match.score;                              // [0, 1] from Vectorize cosine
  const tier = match.metadata.quality_tier;
  const tierBoost = tier === 'featured' ? 0.20 :
                    tier === 'solid'    ? 0.10 : 0;
  const typeBoost = entityTypeRelevanceBoost(match.metadata.entity_type, query);
  return semantic * 0.70 + tierBoost + typeBoost;
}

function entityTypeRelevanceBoost(entityType, query) {
  const q = query.toLowerCase();
  if (entityType === 'plugin'      && /\bplugin\b/.test(q))                       return 0.05;
  if (entityType === 'mcp_server'  && /\b(mcp|model context|server)\b/.test(q))   return 0.05;
  if (entityType === 'skill'       && /\bskill\b/.test(q))                        return 0.02;  // smaller because skill is the implicit default
  return 0;
}
```

**Why this formula:**
- Semantic similarity dominates (0.70 weight) — relevance always wins.
- Tier is a per-type-normalized signal (a "featured" plugin and a "featured" skill
  are both top-of-their-class; the boost is the same).
- Type relevance only fires on explicit type-naming queries (rare); doesn't
  penalize cross-type results.
- Caps cleanly at `1.00 = 0.70 (perfect semantic) + 0.20 (featured) + 0.05 (typed query) + epsilon`.

**Calibration plan (3.5 input, not F2 deliverable):**
- D1 search-query log already captures every query (`worker/index.js:294-`).
- After 3.2 ships and ~30 days of mixed-type query logs accumulate, sample 100
  queries and have a human label top-3 results as relevant/irrelevant.
- Tune weights (potentially per-type tier boosts) against the labeled set.

In F2, the formula is implemented but only exercises the skill branch. **No A/B
or shadow-traffic is needed because the formula reduces to identical ordering
when all results share entity_type='skill'** (semantic dominates, tier boost
applies uniformly, type boost = 0 for skill queries that don't say "skill").

**The upcaster (`scripts/lib/legacy-skill-reader.js`):**

```javascript
// scripts/lib/legacy-skill-reader.js
import { VERSION_HEADER_KEY, CURRENT_SCHEMA_VERSION } from './entity-version.js';

const UPCASTERS = {
  1: upcastV1ToV2,   // pre-F2 shape → EntityRecord<SkillExtra>
};

export function upcastRecord(raw) {
  // schema_version present and recognized → return as-is
  if (raw.schema_version === CURRENT_SCHEMA_VERSION) return raw;
  if (raw.schema_version && !UPCASTERS[raw.schema_version]) {
    throw new Error(`Refusing to load record with unknown schema_version=${raw.schema_version}`);
  }
  const fromVersion = raw.schema_version ?? 1;
  return UPCASTERS[fromVersion](raw);
}

function upcastV1ToV2(v1) {

    name: v1.name,
    description: v1.description,

    repo_url: v1.repo_url,
    repo_stars: v1.repo_stars,
    repo_forks: v1.repo_forks,
    repo_open_issues: v1.repo_open_issues,
    repo_topics: v1.repo_topics ?? [],
    repo_license: v1.repo_license ?? null,
    repo_language: v1.repo_language ?? null,
    repo_created_at: v1.repo_created_at,
    repo_updated_at: v1.repo_updated_at,
```

**Schema-version header line on NDJSON files:**

F1 already supports NDJSON. F2 establishes the convention that the **first
record** of any `data/*.ndjson` file is a header object:

```json

```

The streaming reader `readNdjsonRecords()` (from F1) gets a small extension: if
the first line has `_header: true`, it's consumed as a metadata sentinel and not
yielded as a record. `loadEntities()` reads the header, verifies
`schema_version === 2` (or routes to upcaster if v1), and proceeds. **F1 needs to
land this header convention** — small dependency, called out as an open question
to the F1 planner if not already in their scope.
