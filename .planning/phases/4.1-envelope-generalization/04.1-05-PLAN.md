```js
import { scoreStars, scoreLicense, scoreDescription } from '../../score.js';
export const FEED_WEIGHTS = Object.freeze({ access: 0.20, reachability: 0.15, docs: 0.15, license: 0.15, freshness: 0.15, popularity: 0.15, adoption: 0.05 });
export const NEUTRAL_REACHABILITY = 70;   // spec: "starts neutral at 0.7 until 7 days of history"
export const NEUTRAL_POPULARITY = 50;
/** docs_url present 40 + machine-readable spec/schema present 40 + description quality 0-20 */
export function scoreFeedDocs(record) {
  const e = record?.extra || {};
  let s = 0;
  if (typeof record?.docs_url === 'string' && record.docs_url) s += 40;
  if ((typeof e.spec_url === 'string' && e.spec_url) || (typeof e.schema_url === 'string' && e.schema_url)) s += 40;
  s += scoreDescription(record?.description || '') * 0.2;
  return Math.min(100, s);
}
/** declared-any 70 / permissive 100 / undeclared 0 — scoreLicense's existing ladder IS the spec's three steps */
export function scoreFeedLicense(record) { const l = record?.access?.license || {}; return scoreLicense(l.spdx ?? l.name ?? null); }
/** dataset Freshness: 365-day cliff (api uses score.js scoreRecency, 180-day cliff) */
export function scoreDatasetFreshness(iso) {
  if (!iso) return 0;
  const t = new Date(iso).getTime(); if (!Number.isFinite(t)) return 0;
  const days = (Date.now() - t) / 86400000;
  if (days <= 30) return 100; if (days <= 90) return 90; if (days <= 180) return 75; if (days <= 365) return 50; if (days <= 730) return 25; return 5;
}
export function feedFreshnessInput(record) { return record?.extra?.source_modified_at ?? record?.repo_pushed_at ?? null; }
export function scoreFeedPopularity(record) { return typeof record?.repo_stars === 'number' ? scoreStars(record.repo_stars) : NEUTRAL_POPULARITY; }
export function scoreAdoption(record) { const cb = record?.extra?.consumed_by || {}; const n = (cb.skills?.length || 0) + (cb.plugins?.length || 0) + (cb.mcp_servers?.length || 0); return Math.min(100, n * 20); }
export function weightedTotal(scores, weights) { let t = 0; for (const [k, w] of Object.entries(weights)) t += (scores[k] || 0) * w; return Math.round(Math.min(100, Math.max(0, t))); }
```

/** Score an api EntityRecord. SIDE-EFFECT: writes record.extra.access_clarity (0-1). @returns 0-100 integer */
export function scoreApi(record) {
  const clarity = accessClarity(record?.access);
  if (record?.extra && typeof record.extra === 'object') record.extra.access_clarity = clarity;
  const scores = {
    access: clarity * 100,
    reachability: NEUTRAL_REACHABILITY,          // Phase 4.4 probe history replaces this
    docs: scoreFeedDocs(record),
    license: scoreFeedLicense(record),

    popularity: scoreFeedPopularity(record),
    adoption: scoreAdoption(record),             // Phase 4.6 consumer graph feeds consumed_by
  };
  return weightedTotal(scores, FEED_WEIGHTS);
}

export function buildApiEmbeddingInput(rec) {
  const e = rec?.extra || {};
  const cats = Array.isArray(e.categories_src) ? e.categories_src.filter((c) => typeof c === 'string') : [];
  const parts = [rec?.name || '', typeof e.provider === 'string' ? e.provider : '', rec?.description || '', cats.join(', ')];
  if (Array.isArray(e.endpoint_summaries) && e.endpoint_summaries.length) {
    parts.push(e.endpoint_summaries.filter((s) => typeof s === 'string').slice(0, 10).join('; '));
  }
  return parts.join(' | ').slice(0, 300);
}

export function buildDatasetEmbeddingInput(rec) {
  const e = rec?.extra || {};
  const tags = (Array.isArray(rec?.tags) ? rec.tags : []).filter((t) => typeof t === 'string').map((t) => (t.startsWith('category:') ? t.slice('category:'.length) : t));

  return [rec?.name || '', typeof e.publisher === 'string' ? e.publisher : '', rec?.description || '', tags.join(', '), formats.join(', ')].join(' | ').slice(0, 300);
}
